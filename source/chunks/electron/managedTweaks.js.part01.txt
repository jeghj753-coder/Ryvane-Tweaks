const { spawn, execFile } = require('child_process');
const fs = require('fs');
const path = require('path');
const os = require('os');

const ROOT_DIR = process.platform === 'win32'
  ? path.join(process.env.ProgramData || 'C:\\ProgramData', 'Ryvane', 'ManagedTweaks')
  : path.join(os.tmpdir(), 'Ryvane', 'ManagedTweaks');
const BACKUP_DIR = path.join(ROOT_DIR, 'Backups');

function ensureDir(dir) { try { fs.mkdirSync(dir, { recursive: true }); } catch {} }
function safeId(id) { return String(id || '').replace(/[^a-zA-Z0-9_-]/g, '_').slice(0, 96); }
function psQuote(s) { return String(s ?? '').replace(/'/g, "''"); }
function makeTempName(prefix, ext) {
  return path.join(os.tmpdir(), `${prefix}_${process.pid}_${Date.now()}_${Math.random().toString(36).slice(2,8)}${ext}`);
}
function killProcessTree(pid) {
  if (!pid || process.platform !== 'win32') return;
  try { execFile('taskkill.exe', ['/PID', String(pid), '/T', '/F'], { windowsHide: true }, () => {}); } catch {}
}
function runChild(file, args, timeout = 60000) {
  return new Promise(resolve => {
    let stdout = '', stderr = '', settled = false, timedOut = false;
    let child;
    try {
      child = spawn(file, args, { windowsHide: true, shell: false, stdio: ['ignore', 'pipe', 'pipe'] });
    } catch (e) { return resolve({ success: false, output: e?.message || String(e) }); }
    child.stdout?.on('data', d => stdout += d.toString());
    child.stderr?.on('data', d => stderr += d.toString());
    const timer = setTimeout(() => { timedOut = true; killProcessTree(child.pid); try { child.kill('SIGKILL'); } catch {} }, timeout);
    child.on('error', e => { if (settled) return; settled = true; clearTimeout(timer); resolve({ success: false, output: e?.message || String(e), stdout, stderr }); });
    child.on('close', code => { if (settled) return; settled = true; clearTimeout(timer); resolve({ success: !timedOut && code === 0, output: (stdout || stderr || '').trim(), stdout, stderr, code, timedOut }); });
  });
}
function runPS(script, timeout = 60000) {
  const file = makeTempName('ts_managed', '.ps1');
  const body = `Set-ExecutionPolicy -Scope Process -ExecutionPolicy Bypass -Force\r\n$ProgressPreference='SilentlyContinue'\r\n$ErrorActionPreference='SilentlyContinue'\r\n${script}\r\n`;
  fs.writeFileSync(file, Buffer.concat([Buffer.from([0xef,0xbb,0xbf]), Buffer.from(body, 'utf8')]));
  return runChild('powershell.exe', ['-NoProfile','-NonInteractive','-ExecutionPolicy','Bypass','-WindowStyle','Hidden','-File',file], timeout)
    .then(res => { try { fs.unlinkSync(file); } catch {} return res; });
}

async function runFirmwareCompatibilityScan() {
  const script = `
$ErrorActionPreference='SilentlyContinue'
function YesNo($v){ if($null -eq $v){ return 'Unknown' }; if($v){ return 'Yes' } else { return 'No' } }
$cpu = Get-CimInstance Win32_Processor | Select-Object -First 1 Name,Manufacturer,NumberOfCores,NumberOfLogicalProcessors,VirtualizationFirmwareEnabled
$board = Get-CimInstance Win32_BaseBoard | Select-Object -First 1 Manufacturer,Product,Version,SerialNumber
$bios = Get-CimInstance Win32_BIOS | Select-Object -First 1 Manufacturer,SMBIOSBIOSVersion,ReleaseDate
$cs = Get-CimInstance Win32_ComputerSystem | Select-Object -First 1 Manufacturer,Model,SystemType,HypervisorPresent
$fwType = 'Unknown'
try {
  $ci = Get-ComputerInfo -Property BiosFirmwareType -EA Stop
  if($ci.BiosFirmwareType){ $fwType = [string]$ci.BiosFirmwareType }
} catch {}
if($fwType -eq 'Unknown' -or [string]::IsNullOrWhiteSpace($fwType)){
  try {
    $fwRaw = (Get-ItemProperty -Path 'HKLM:\\SYSTEM\\CurrentControlSet\\Control' -Name PEFirmwareType -EA Stop).PEFirmwareType
    if($fwRaw -eq 2){ $fwType = 'UEFI' } elseif($fwRaw -eq 1){ $fwType = 'Legacy BIOS' }
  } catch {}
}
$secureBoot = 'Unavailable'
try {
  $sb = Confirm-SecureBootUEFI -EA Stop
  $secureBoot = if($sb){ 'Enabled' } else { 'Disabled' }
  if($fwType -eq 'Unknown'){ $fwType = 'UEFI' }
} catch {
  $msg = $_.Exception.Message
  if($msg -match 'not supported|not support|Cmdlet not supported'){ if($fwType -eq 'Unknown'){ $fwType = 'Legacy BIOS' } }
  $secureBoot = 'Unavailable'
}
$deviceGuard = Get-CimInstance -Namespace root\\Microsoft\\Windows\\DeviceGuard -ClassName Win32_DeviceGuard -EA SilentlyContinue
$bcd = (& bcdedit.exe /enum '{current}' 2>&1) -join [Environment]::NewLine
$bitlocker = 'Unknown'
try { $bl = Get-BitLockerVolume -MountPoint $env:SystemDrive -EA Stop; $bitlocker = $bl.ProtectionStatus.ToString() } catch {}
$firmwareAccess = if($fwType -eq 'UEFI') { 'Readable check only - vendor write profile required for BIOS tuning' } else { 'Not available on Legacy BIOS boot' }
$supportedWindowsControls = @(
  @{ id='ts-elam-toggle'; name='ELAM boot option'; support='Supported by BCD when accepted by this Windows build'; reversible='Backup/deletevalue restore'; restart=$true },
  @{ id='ts-isolated-context-toggle'; name='Isolated context boot option'; support='Attempted through BCD; unsupported builds will be blocked'; reversible='Backup/deletevalue restore'; restart=$true },
  @{ id='ts-inmemory-kernel-toggle'; name='Kernel mitigation option'; support='Attempted through BCD; unsupported builds will be blocked'; reversible='Backup/deletevalue restore'; restart=$true },
  @{ id='ts-hypervisor-off'; name='Hypervisor launch'; support='Supported BCD option'; reversible='Backup/deletevalue restore'; restart=$true },
  @{ id='ts-vbs-off'; name='VBS policy/BCD'; support='Windows policy + BCD where available'; reversible='Registry/BCD backup restore'; restart=$true }
)
$firmwareControls = @(
  @{ id='ts-bios-cstates'; name='CPU C-States'; status='Vendor profile required'; canToggle=$false; reason='Motherboard-specific BIOS option, not a universal Windows value.' },
  @{ id='ts-bios-package-cstate'; name='Package C-State Limit'; status='Vendor profile required'; canToggle=$false; reason='Motherboard-specific BIOS option.' },
  @{ id='ts-bios-c1e'; name='C1E Support'; status='Vendor profile required'; canToggle=$false; reason='Motherboard-specific BIOS option.' },
  @{ id='ts-bios-pcie-clock-gating'; name='PCIe Clock Gating'; status='Vendor profile required'; canToggle=$false; reason='Motherboard-specific firmware option.' },
  @{ id='ts-bios-energy-efficient-pstate'; name='Energy Efficient P-State'; status='Vendor profile required'; canToggle=$false; reason='Motherboard-specific firmware option.' },
  @{ id='ts-bios-aspm'; name='ASPM'; status='Windows-side alternative available'; canToggle=$false; reason='Firmware ASPM is BIOS-specific. Windows PCIe Link State is handled in Power settings.' },
  @{ id='ts-bios-ltr'; name='LTR'; status='Vendor profile required'; canToggle=$false; reason='Motherboard-specific firmware option.' },
  @{ id='ts-bios-race-to-halt'; name='Race To Halt'; status='Vendor profile required'; canToggle=$false; reason='Motherboard-specific firmware option.' },
  @{ id='ts-bios-xhci-handoff'; name='XHCI Hand-off'; status='Vendor profile required'; canToggle=$false; reason='Motherboard-specific BIOS option.' },
  @{ id='ts-bios-virtualization'; name='VT-x / SVM firmware virtualization'; status='Firmware only'; canToggle=$false; reason='Windows can disable Hypervisor/VBS, but CPU virtualization itself is controlled in firmware.' },
  @{ id='ts-bios-pci-delay'; name='PCI Delay Optimization'; status='Vendor profile required'; canToggle=$false; reason='Not a universal Windows value.' }
)
$result = [ordered]@{
  success = $true
  scannedAt = (Get-Date).ToString('yyyy-MM-dd HH:mm:ss')
  firmwareType = $fwType
  secureBoot = $secureBoot
  firmwareAccess = $firmwareAccess
  bitLocker = $bitlocker
  cpu = [ordered]@{ name=$cpu.Name; manufacturer=$cpu.Manufacturer; cores=$cpu.NumberOfCores; threads=$cpu.NumberOfLogicalProcessors; virtualizationFirmwareEnabled=(YesNo $cpu.VirtualizationFirmwareEnabled) }
  motherboard = [ordered]@{ manufacturer=$board.Manufacturer; product=$board.Product; version=$board.Version; system=$cs.Model }
  bios = [ordered]@{ manufacturer=$bios.Manufacturer; version=$bios.SMBIOSBIOSVersion; releaseDate=$bios.ReleaseDate }
  hypervisorPresent = (YesNo $cs.HypervisorPresent)
  bcdDetected = ($bcd.Length -gt 0)
  windowsControls = $supportedWindowsControls
  firmwareControls = $firmwareControls
  summary = if($fwType -eq 'UEFI') { 'UEFI detected. Windows-level boot/security controls can be managed safely with backup. Direct BIOS tuning requires a supported vendor firmware profile.' } else { 'Legacy BIOS boot detected. UEFI firmware variables are not available from Windows.' }
}
$result | ConvertTo-Json -Depth 8
`;
  const res = await runPS(script, 45000);
  if (!res.success) return { success: false, message: res.output || 'Firmware scan failed.' };
  const raw = String(res.output || '').trim();
  try { return JSON.parse(raw); } catch {}
  const start = raw.indexOf('{');
  const end = raw.lastIndexOf('}');
  if (start >= 0 && end > start) {
    try { return JSON.parse(raw.slice(start, end + 1)); } catch {}
  }
  return { success: false, message: 'Firmware scan returned unreadable data.', raw };
}
