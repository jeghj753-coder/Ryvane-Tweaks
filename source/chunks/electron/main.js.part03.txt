      return {
        ...data,
        message: data.message || 'License deactivation failed. Your local license was kept so you can retry.',
      }
    }

    licenseStore.delete('licenseStatus')
    return data
  } catch (err) {
    console.error('[License Deactivate Error]', {
      code: err?.code,
      message: err?.message,
      status: err?.response?.status,
      data: err?.response?.data,
    })
    if (err.response?.data) return { ...err.response.data, localCleared: false }
    return { success: false, localCleared: false, message: 'Could not reach deactivation server. License was not removed, so the key stays safe on this PC.' }
  }
})

ipcMain.handle('verify-license', async (_, licenseKey) => {
  try {
    const axios = require('axios')
    const cleanKey = String(licenseKey || '').trim()
    if (!cleanKey) return { success: false, message: 'Please enter a license key.' }

    const deviceId = getDeviceId()
    const response = await axios.post(
      `${LICENSE_SERVER_URL}/api/verify-license`,
      { licenseKey: cleanKey, deviceId, deviceName: os.hostname(), appVersion: app.getVersion() },
      { timeout: 70000 }
    )

    const data = response.data || {}
    if (!data.success) return data

    const stored = buildStoredLicense(data, cleanKey)
    licenseStore.set('licenseStatus', stored)
    return { ...stored, success: true }
  } catch (err) {
    console.error('[License Verify Error]', {
      code: err?.code,
      message: err?.message,
      status: err?.response?.status,
      data: err?.response?.data,
    })
    if (err.response?.data) return err.response.data
    return { success: false, message: 'Could not reach activation server. Check your internet connection.' }
  }
})
