  adminStatusCached = !!adminMode

  return new Promise((resolve) => {
    mainWindow = new BrowserWindow({
      width: 1440,
      height: 820,
      minWidth: 1200,
      minHeight: 700,
      frame: false,
      backgroundColor: '#0d0d0f',  // matches --bg-base, prevents white flash
      icon: fs.existsSync(path.join(__dirname, 'icon.ico')) ? path.join(__dirname, 'icon.ico') : (fs.existsSync(path.join(__dirname, '../build/icon.ico')) ? path.join(__dirname, '../build/icon.ico') : undefined),
      webPreferences: {
        preload: path.join(__dirname, 'preload.js'),
        contextIsolation: true,
        nodeIntegration: false,
        backgroundThrottling: true,    // performance: let Electron throttle when minimized/backgrounded
        spellcheck: false,
        enableWebSQL: false,
      },
      show: false,
      paintWhenInitiallyHidden: true,  // start painting before show()
    })

    if (isDev) {
      mainWindow.loadURL('http://localhost:3000')
    } else {
      mainWindow.loadFile(path.join(app.getAppPath(), 'build', 'index.html'))
    }

    // Resolve only after first paint is ready — eliminates blink during splash handoff.
    mainWindow.once('ready-to-show', () => {
      mainWindow?.webContents.send('admin-status', adminMode)
      if (pendingToolOAuthCode) {
        mainWindow?.webContents.send('tool-oauth-code', pendingToolOAuthCode)
        pendingToolOAuthCode = null
      }
      if (showOnReady) {
        setTimeout(() => {
          mainWindow?.show()
          resolve(mainWindow)
        }, 120)
      } else {
        mainWindow?.setOpacity?.(0)
        resolve(mainWindow)
      }
    })

    mainWindow.on('closed', () => { mainWindow = null })
  })
}

// ─── Auto Updater ─────────────────────────────────────────
function setupAutoUpdater() {
  if (isDev) return

  autoUpdater.autoDownload = false
  autoUpdater.autoInstallOnAppQuit = true
  autoUpdater.allowPrerelease = false
  autoUpdater.allowDowngrade = false
  // Important: update access is not tied to Free/Premium licensing.
  // Code-signature verification is disabled in package.json because this app is currently unsigned.

  autoUpdater.on('checking-for-update', () => {
    emitUpdateStatus({ status: 'checking', message: 'Checking for updates...' })
  })
  autoUpdater.on('update-available', (info) => {
    emitUpdateStatus({
      status: 'available',
      message: `New update available: v${info?.version || ''}`,
      version: info?.version || null, info
    })
  })
  autoUpdater.on('update-not-available', (info) => {
    emitUpdateStatus({
      status: 'not-available',
      message: 'You are using the latest version.',
      version: info?.version || null
    })
  })
  autoUpdater.on('download-progress', (progress) => {
    emitUpdateStatus(buildDownloadProgressPayload(progress))
  })
  autoUpdater.on('update-downloaded', (info) => {
    emitUpdateStatus({
      status: 'downloaded',
      message: 'Update downloaded. Restart to install.',
      version: info?.version || null
    })
  })
  autoUpdater.on('error', (err) => {
    emitUpdateStatus({
      status: 'error', message: err?.message || 'Update error.'
    })
  })
}

ipcMain.handle('run-tweak', async (_, key, arg) => {
  const action = arg && typeof arg === 'object' ? String(arg.action || 'apply') : 'apply'
  const value = arg && typeof arg === 'object' && Object.prototype.hasOwnProperty.call(arg, 'value') ? arg.value : arg

  if (action === 'revert') {
    const tweakKey = String(key || '')
    const fn = TWEAK_REVERT_MAP[tweakKey]
    if (!fn) {
      // Legacy rows merged into the new pages can use their original data-tweak revert script.
      // This fixes errors like: "No revert action registered for tweak: fast-shutdown".
      if (DATA_TWEAKS && DATA_TWEAKS[tweakKey]?.revertScript && TWEAK_MAP.revertDataTweak) {
        try { return await TWEAK_MAP.revertDataTweak(tweakKey) } catch (e) { return { success: false, output: e.message } }
      }
      return { success: false, output: `No revert action registered for tweak: ${key}` }
    }
    try { return await fn(value) } catch (e) { return { success: false, output: e.message } }
  }

  if (isPremiumRequiredForTweak(key, value) && !hasPremiumAccess()) {
    const message = 'Ryvane Tweaks license required for this tweak.'
    return { success: false, locked: true, message, output: message }
  }

  const tweakKey = String(key || '')
  const fn = TWEAK_MAP[tweakKey]
  if (!fn) {
    // Direct legacy data-tweak IDs are supported so old merged rows remain fully apply/revert capable.
    if (DATA_TWEAKS && DATA_TWEAKS[tweakKey]?.script && TWEAK_MAP.applyDataTweak) {
      try { return await TWEAK_MAP.applyDataTweak(tweakKey) } catch (e) { return { success: false, output: e.message } }
    }
    return { success: false, output: `Unknown: ${key}` }
  }
  try { return await fn(value) } catch (e) { return { success: false, output: e.message } }
})

function getHardwareProfilePath() {
  return path.join(app.getPath('userData'), 'tweakshift-hardware-profile.json')
}

function readHardwareProfileCache() {
  try {
    const fp = getHardwareProfilePath()
    if (!fs.existsSync(fp)) return null
    const cached = JSON.parse(fs.readFileSync(fp, 'utf8'))
    if (!cached || typeof cached !== 'object') return null
    return cached
  } catch { return null }
}

function saveHardwareProfileCache(profile) {
  try { fs.writeFileSync(getHardwareProfilePath(), JSON.stringify(profile, null, 2), 'utf8') } catch {}
}

async function scanHardwareProfileOnce(force = false) {
  if (!force) {
    const cached = readHardwareProfileCache()
    if (cached?.cpu || cached?.gpu || cached?.ram) return { ...cached, cached: true }
  }

  const cpu = os.cpus()?.[0]?.model || 'Unknown CPU'
  const ramBytes = os.totalmem() || 0
  const ramGbRaw = ramBytes / 1073741824
  const ram = ramBytes ? `${Math.round(ramGbRaw)} GB RAM` : 'Unknown RAM'
  let gpu = 'Unknown GPU'

  try {
    if (si?.graphics) {
      const gpuData = await si.graphics()
      const controllers = Array.isArray(gpuData?.controllers) ? gpuData.controllers : []
      const selected = controllers.find(g => /nvidia|geforce|rtx|gtx|amd|radeon|intel/i.test(`${g.vendor || ''} ${g.model || ''}`)) || controllers[0]
      if (selected) gpu = selected.model || selected.vendor || gpu
    }
  } catch {}

  const profile = {
    system: `${os.type()} ${os.release()}`.trim(),
    cpu,
    gpu,
    ram,
    ramBytes,
    scannedAt: new Date().toISOString(),
    cached: false
  }

  saveHardwareProfileCache(profile)
  return profile
}

ipcMain.handle('deactivate-license', async () => {
  try {
    const axios = require('axios')
    const current = licenseStore.get('licenseStatus') || {}
    const licenseKey = current.licenseKey || current.license_key || ''
    const deviceId = getDeviceId()

    if (!licenseKey) {
      licenseStore.delete('licenseStatus')
      return { success: true, message: 'License removed locally.' }
    }

    const response = await axios.post(
      `${LICENSE_SERVER_URL}/api/deactivate-license`,
      {
        licenseKey,
        deviceId,
        source: current.source || current.provider || '',
        freemiusUid: current.freemiusUid || current.uid || '',
        freemiusInstallId: current.freemiusInstallId || current.installId || current.install_id || '',
        freemiusInstallApiToken: current.freemiusInstallApiToken || current.installApiToken || current.install_api_token || '',
        freemiusLicenseId: current.freemiusLicenseId || current.licenseId || current.license_id || '',
      },
      { timeout: 70000 }
    )

    const data = response.data || { success: true, message: 'License deactivated.' }
    if (data.success === false) {
