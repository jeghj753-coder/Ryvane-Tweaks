function makeRegistryPS(tweak, backupPath, apply = true) {
  const entriesJson = JSON.stringify(tweak.entries || []);
  return `
$backupPath='${psQuote(backupPath)}'
$entries = '${psQuote(entriesJson)}' | ConvertFrom-Json
function To-PSPath([string]$hive,[string]$sub) { if ($hive -eq 'HKLM') { return 'HKLM:\\' + $sub } else { return 'HKCU:\\' + $sub } }
function Get-RegValue([string]$hive,[string]$sub,[string]$name) {
  $p = To-PSPath $hive $sub
  $res = [ordered]@{ hive=$hive; path=$sub; name=$name; keyExists=(Test-Path $p); valueExists=$false; type=$null; data=$null }
  if (Test-Path $p) { try { $it = Get-ItemProperty -Path $p -Name $name -EA Stop; $res.valueExists=$true; $res.data=$it.$name } catch {} }
  return $res
}
if (${apply ? '$true' : '$false'}) {
  $backup = @()
  foreach($e in $entries) { $backup += Get-RegValue $e[0] $e[1] $e[2] }
  New-Item -ItemType Directory -Path (Split-Path $backupPath) -Force | Out-Null
  $backup | ConvertTo-Json -Depth 6 | Set-Content -Path $backupPath -Encoding UTF8
  ${tweak.actionBefore || ''}
  foreach($e in $entries) {
    $p = To-PSPath $e[0] $e[1]
    New-Item -Path $p -Force | Out-Null
    $name = [string]$e[2]; $type = [string]$e[3]; $data = $e[4]
    if ($name -eq '(default)') { $name = '(default)' }
    try { New-ItemProperty -Path $p -Name $name -Value $data -PropertyType ($type -replace 'REG_','') -Force | Out-Null } catch { Set-ItemProperty -Path $p -Name $name -Value $data -Force }
  }
  Write-Output 'Applied and backed up registry values.'
} else {
  if (!(Test-Path $backupPath)) { Write-Output 'No backup found for this tweak.'; exit 2 }
  $backup = Get-Content $backupPath -Raw | ConvertFrom-Json
  foreach($b in $backup) {
    $p = To-PSPath $b.hive $b.path
    if ($b.valueExists -eq $true) {
      New-Item -Path $p -Force | Out-Null
      try { Set-ItemProperty -Path $p -Name $b.name -Value $b.data -Force } catch {}
    } else {
      if (Test-Path $p) { try { Remove-ItemProperty -Path $p -Name $b.name -Force -EA SilentlyContinue } catch {} }
    }
  }
  Write-Output 'Restored registry values from backup.'
}`;
}
